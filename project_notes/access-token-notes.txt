 with user_follow_page_cte as 

(

select 
       
      uf.user_id,
      uf.friend_id,
      up.page_id as friends_page_id
from 
     users_friends uf
left join
          users_pages up on uf.friend_id = up.user_id 
)

select * from user_follow_page_cte order by 1, 2 













WITH bc AS (
    SELECT
        user_id,
        page_id
    FROM
        users_pages
)

SELECT DISTINCT
    ab.user_id,
    ab.friend_page_id

FROM
    (
        SELECT
            uf.user_id,
            uf.friend_id,
            up.page_id AS friend_page_id

        FROM users_friends AS uf
        LEFT JOIN
            users_pages AS up
            ON uf.friend_id = up.user_id
        ORDER BY uf.user_id, uf.friend_id
    ) AS ab

LEFT JOIN
    bc

    ON ab.user_id = bc.user_id AND ab.friend_page_id = bc.page_id
WHERE
    bc.page_id IS NULL AND ab.friend_page_id IS NOT NULL
